import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ActivatedRoute, Router } from '@angular/router';
import { FormsModule } from '@angular/forms';
import { InsuranceService } from '../service/insurance.service';
import { Plan } from '../model/plan';

@Component({
  selector: 'app-questions',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './questions.html'
})
export class Questions implements OnInit {

  // 🔹 Plan
  plan!: Plan;

  // 🔹 User inputs
  name!: string;
  city!: string;
  phone!: string;
  email!: string;
  age!: number;
  income!: number;

  disease = false;
  smoker = false;

  // 🔹 Calculated premium
  premiumAmount = 0;

  constructor(
    private route: ActivatedRoute,
    private router: Router,
    private insuranceService: InsuranceService
  ) {}

  ngOnInit(): void {
    const planId = Number(this.route.snapshot.paramMap.get('planId'));

    this.insuranceService.getPlanById(planId).subscribe(plans => {
      this.plan = plans[0];
    });
  }

  calculatePremium(): void {
    if (!this.plan || !this.age || !this.income) return;

    let amount = this.plan.baseAmt;

    if (this.age > 45) amount += amount * 0.20;
    if (this.income > 1000000) amount += amount * 0.10;
    if (this.disease) amount += amount * 0.30;
    if (this.smoker) amount += amount * 0.25;

    this.premiumAmount = Math.round(amount);
  }

  proceedToPayment(): void {
    const bookingData = {
      name: this.name,
      city: this.city,
      phone: this.phone,
      email: this.email,
      age: this.age,
      planId: this.plan.planId,
      planName: this.plan.planName,
      validity: this.plan.validity,
      premiumAmt: this.premiumAmount
    };

    console.log('Booking Data:', bookingData);

    this.router.navigate(['/payment'], {
      state: { bookingData }
    });
  }
}
